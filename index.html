Firebase 보안 규칙 · 대규모 확장 설정 가이드

1. Firebase Console 필수 활성화 항목:

  • Authentication → Sign-in method: 이메일/비밀번호 사용 설정
  • Storage: 버킷 생성 (썸네일 이미지 분리 저장으로 1MiB 한계 영구 해결)
  • Firestore 복합 인덱스: 카테고리별 정렬 쿼리 최초 실행 시 콘솔 로그에 출력되는 링크를 클릭하면 자동 생성됩니다.
Firestore Security Rules (복사하여 붙여넣기) 
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isSignedIn() {
      return request.auth != null && request.auth.token.firebase.sign_in_provider != 'anonymous';
    }

    match /artifacts/{appId}/public/data/projects/{projectId} {
      allow read: if true;

      // 작품 신규 등록: 로그인 사용자만 가능, 문서 ID = entryProjectId 강제, 소유자 UID 일치 필수
      allow create: if isSignedIn()
        && request.resource.data.ownerUid == request.auth.uid
        && request.resource.data.entryProjectId == projectId
        && request.resource.data.title is string
        && request.resource.data.title.size() >= 1
        && request.resource.data.title.size() <= 80;

      // 작품 수정: 소유자는 author/searchKeywords 수정 가능, 타인은 평점 집계 필드만 트랜잭션 업데이트 허용
      allow update: if isSignedIn() && (
        resource.data.ownerUid == request.auth.uid
        || request.resource.data.diff(resource.data).affectedKeys()
            .hasOnly(['reviewCount', 'blindedCount', 'scoreSum', 'avgScore', 'posCount', 'mixCount', 'negCount', 'trendingScore'])
      );

      // 작품 삭제: 오직 등록자 본인(ownerUid)만 허용
      allow delete: if isSignedIn() && resource.data.ownerUid == request.auth.uid;

      // 리뷰 하위 컬렉션 (1MiB 문서 한계 해결 + 계정당 1회 리뷰 강제)
      match /reviews/{reviewUid} {
        allow read: if true;
        allow create: if isSignedIn()
          && reviewUid == request.auth.uid
          && request.resource.data.authorUid == request.auth.uid
          && request.resource.data.score is int
          && request.resource.data.score >= 0 && request.resource.data.score <= 100;
        // 신고 처리 업데이트 (reports, reportCount, aiStatus, aiReason만 변경 가능)
        allow update: if isSignedIn()
          && request.resource.data.diff(resource.data).affectedKeys()
              .hasOnly(['reports', 'reportCount', 'aiStatus', 'aiReason']);
        // 작품 삭제 시 작품 소유자 또는 리뷰 작성자 본인 삭제 허용
        allow delete: if isSignedIn() && (
          reviewUid == request.auth.uid
          || get(/databases/$(database)/documents/artifacts/$(appId)/public/data/projects/$(projectId)).data.ownerUid == request.auth.uid
        );
      }

      // 기기당 1회 평가 잠금 문서
      match /deviceLocks/{deviceHash} {
        allow read: if true;
        allow create: if isSignedIn() && request.resource.data.uid == request.auth.uid;
        allow delete: if isSignedIn() && (
          resource.data.uid == request.auth.uid
          || get(/databases/$(database)/documents/artifacts/$(appId)/public/data/projects/$(projectId)).data.ownerUid == request.auth.uid
        );
      }
    }

    // 공개 프로필 (비밀번호/해시 일체 미저장)
    match /artifacts/{appId}/public/data/profiles/{uid} {
      allow read: if true;
      allow write: if isSignedIn() && request.auth.uid == uid;
    }
  }
}
Firebase Storage Rules (썸네일 업로드) 
rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /artifacts/{appId}/thumbnails/{fileName} {
      allow read: if true;
      allow write: if request.auth != null
        && request.auth.token.firebase.sign_in_provider != 'anonymous'
        && request.resource.size < 1 * 1024 * 1024
        && request.resource.contentType.matches('image/.*');
      allow delete: if request.auth != null;
    }
  }
}
